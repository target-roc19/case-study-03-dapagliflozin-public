libname raw_data " ";

data egfr_code;
	infile "F:\DataManagement\RenalFunctionMacro-Aurum\egfr_AD_2022.txt" dsd dlm='09'x firstobs=2 truncover;
	informat 	MedcodeId $19.;
	format 		MedcodeId $19.;
	input MedcodeId $;
run;

proc sort data = egfr_code;
by medcodeid;
run;

proc sort data = raw_data.observation;
by medcodeid;
run;

data egfr; 
merge egfr_code (in = a)  raw_data.observation (in = b);
by medcodeid;
if a and b;
if obsdate = . then delete;
if value = . then delete;
run;

proc freq data = egfr;
table numunitid;
run;

proc univariate data = egfr;
var value;
run;

data egfr2;
set egfr;
if value >=1 and value <= 180;
run;

*99% has numunit 285, umol/L;

proc freq data = creat;
tables numunitid;
run;


data creat1;
	set creat;
	if numunitid in (191,197,218,285,816);
	/* convert to umol/L */
	if value = 218 then value = (value * 1000);		/* mmol/L */
	if value = 816 then value = (value * 1000000);		/* mol/L */
	if value = 285 then value = value;			/* umol/L */
	if value = 191 then value = value;			/* umol/L */
	if value = 197 then value = value;			/* umol/L */
	if value > 0;
run;

proc sort data = creat1;
by patid;
run;

proc sort data = raw_data.patient;
by patid;
run;

data creat2;
merge creat1 (in = a) raw_data.patient(keep = patid gender yob in = b);
by patid;
if a and b;
run;

*Few observations deleted, due to invalid obsdate resulting in a negative age;
data creat3;
set creat2;
age = year (obsdate) - yob;
sex = gender - 1; 
label sex = "sex, 0 = men, 1 = women";

/* formule uses mg/dL --> dus /88.4 */
if sex = 1 then gfr = (186)*((value/88.4)**-1.154)*(age**-0.203)*(0.742); *for women;
	else gfr = (186)*((value/88.4)**-1.154)*(age**-0.203); *for men;
	if 1 =< gfr =< 180;
;
run;


data mdrd;
	set egfr2 (keep=patid obsdate value rename=value=gfr) creat3 (keep=patid obsdate gfr);
run;

proc sort data = mdrd;
by patid obsdate;
run;

data mdrd2;
set mdrd;
run;

*All double GFR values on the same day deleted;
proc sort data = mdrd2 nodupkey;
by patid obsdate gfr;
run;


proc sort data = mdrd2 ;
by patid obsdate  descending gfr ;
run;


data mdrd3;
set mdrd2;
by patid obsdate;
if first.obsdate;
run;


data egfr;
set mdrd3;

ckd1 = 0; ckd2 = 0;  ckd3 = 0;  ckd4 =  0; ckd5 = 0; 

	if  0  < gfr < 15 then ckd5 = 1;
   	else if 15 =< gfr < 30 then ckd4 = 1;
	else if 30 =< gfr < 60 then ckd3 = 1;
	else if 60 =< gfr < 90 then ckd2 = 1;
	else if 90 =< gfr  then ckd1 = 1;
run;

proc freq data = egfr;
tables ckd1 ckd2 ckd3 ckd4 ckd5;
run;

%macro medical (var= ,code=);

data &var._cod;
	infile "F:\DataManagement\RenalFunctionMacro-Aurum\&code..txt" dsd dlm='09'x firstobs=2 truncover;
	informat 	MedcodeId $19.;
	format 		MedcodeId $19.;
	input MedcodeId $;
run;
proc sort data = &var._cod;
	by medcodeid;
run;
proc sort data = raw_data.observation;
	by medcodeid;
run;
data &var.;
	merge raw_data.observation (in=ina keep=patid medcodeid obsdate) &var._cod (in=inb keep=medcodeid);
	by medcodeid;
	if ina and inb;
run;
%mend medical;

%medical (var=ckd1,code=CKDstage1_AD_may2022);
%medical (var=ckd2,code=CKDstage2_AD_may2022);
%medical (var=ckd3,code=CKDstage3_AD_may2022);
%medical (var=ckd4,code=CKDstage4_AD_may2022);
%medical (var=ckd5,code=CKDstage5_AD_may2022);

data ckdstage;
	set ckd1 (keep=patid obsdate in=ina)
		ckd2 (keep=patid obsdate in=inb)
		ckd3 (keep=patid obsdate in=inc)
		ckd4 (keep=patid obsdate in=ind)
		ckd5 (keep=patid obsdate in=ine);
	if ina then do;
		ckd1 = 1;
		ckd2 = 0; ckd3 = 0; ckd4 = 0; ckd5 = 0; 
	end;
	if inb then do;
		ckd2 = 1;
		ckd1 = 0; ckd3 = 0; ckd4 = 0; ckd5 = 0; 
	end;
	if inc then do;
		ckd3 = 1;
		ckd1 = 0; ckd2 = 0; ckd4 = 0; ckd5 = 0;
	end;
	if ind then do;
		ckd4 = 1;
		ckd1 = 0; ckd2 = 0; ckd3 = 0; ckd5 = 0; 
	end;
	if ine then do;
		ckd5 = 1;
		ckd1 = 0; ckd2 = 0; ckd3 = 0; ckd4 = 0; 
	end;
	ckd_read = 1; 
	if obsdate = . then delete;
run;

*Select best kidney function if multiple records on 1 day;
proc sort data = ckdstage;
	by patid obsdate ckd1 ckd2 ckd3 ckd4 ckd5;
run;

	by patid obsdate ckd1 ckd2 ckd3 ckd4 ckd5;
